--
-- Name: accounts_payable; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.accounts_payable (
    ap_id character varying(15) NOT NULL,
    vendor_id character varying(15),
    invoice_date date,
    due_date date,
    amount numeric,
    status character varying(10),
    currency character varying(3) DEFAULT 'USD'::character varying,
	CONSTRAINT accounts_payable_pkey PRIMARY KEY (ap_id)
);


--
-- Name: accounts_receivable; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.accounts_receivable (
    ar_id character varying(15) NOT NULL,
    customer_id character varying(15),
    invoice_date date,
    due_date date,
    amount numeric,
    status character varying(10),
    currency character varying(3) DEFAULT 'USD'::character varying,
	CONSTRAINT accounts_receivable_pkey PRIMARY KEY (ar_id)
);


--
-- Name: assets; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.assets (
    asset_id character varying(15) NOT NULL,
    asset_name character varying(30),
    puchase_date date,
    cost numeric,
    service_years integer,
    dept character varying(15),
    currency character varying(3) DEFAULT 'USD'::character varying,
	CONSTRAINT assets_pkey PRIMARY KEY (asset_id)
);


--
-- Name: budgets; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.budgets (
    budget_id character varying(15) NOT NULL,
    dept character varying(15),
    year integer,
    amount numeric,
	CONSTRAINT budgets_pkey PRIMARY KEY (budget_id)
);



--
-- Name: chart_of_acct; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.chart_of_acct (
    account integer NOT NULL,
    acct_name character varying(30),
    category character varying(15),
	CONSTRAINT chart_of_acct_pkey PRIMARY KEY (account)
);


--
-- Name: customers; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.customers (
    customer_id character varying(15) NOT NULL,
    customer_name character varying(30),
	CONSTRAINT customers_pkey PRIMARY KEY (customer_id)
);


--
-- Name: depreciation; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.depreciation (
    depreciation_id character varying(10) NOT NULL,
    asset_id character varying(15),
    year integer,
    amount numeric,
    currency character varying(3) DEFAULT 'USD'::character varying,
	CONSTRAINT depreciation_pkey PRIMARY KEY (depreciation_id)
);


--
-- Name: general_ledger; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.general_ledger (
    gl_id character varying(10) NOT NULL,
    date date,
    account integer NOT NULL,
    debit numeric,
    credit numeric,
    description character varying(50),
    currency character varying(3) DEFAULT 'USD'::character varying,
	CONSTRAINT general_ledger_pkey PRIMARY KEY (gl_id, account)
);



--
-- Name: vendors; Type: TABLE; Schema: bookkeeper; Owner: postgres
--

CREATE TABLE bookkeeper.vendors (
    vendor_id character varying(15) NOT NULL,
    vendor_name character varying(30),
	CONSTRAINT vendors_pkey PRIMARY KEY (vendor_id)
);
